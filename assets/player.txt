# Player tuning. Times in seconds (60 Hz sim: 0.15 s = 9 frames).

speed 3.2

turn_speed 16

# Ground movement (Half-Life/Quake shaped): walk 3.2 -> sprint 7.0 (speed * sprint_mult).
# accel/air_accel/friction are Quake's sv_accelerate/sv_friction-shaped coefficients, not m/s^2 --
# see the friction/accel math in player_update (combat.c). jump_height is metres, converted to a
# launch vy against game.c's gravity. crouch_mult only scales speed here; the eye height and stance
# drop are game.c's job.
accel 10
air_accel 12
friction 8
stop_speed 1.4
jump_height 1.0
crouch_mult 0.5

# Traversal (src/combat.c, the "traversal" section). The top speed is not reached by holding Shift,
# it is BUILT: sprint_ramp seconds of running to get from the walk to speed * sprint_mult, and it is
# kept through a hop and spent by stopping. speed_cap is the ceiling nothing gets past, which is
# what keeps bunny hopping fun and finite. air_wish is the m/s the mid-air accel step chases: 0.9 is
# Quake's "steer a little", 1.6 is enough to pick which side of a gap you land on.
sprint_ramp 1.05
speed_cap 8.8
coyote 0.12        # seconds past the edge a jump still counts
jump_buffer 0.15   # seconds a jump press waits for the feet to land
air_wish 1.6

# Slide: Ctrl at a sprint. It bleeds slide_decel m/s every second and ends at slide_exit, or when
# the 0.8 s is up, or when Ctrl comes off -- unless there is a roof, in which case it holds until
# there is headroom, which is what makes a culvert something you go through rather than get stuck in.
slide_time 0.8
slide_decel 1.6
slide_enter 4.6
slide_exit 2.2

# Mantle and vault: no level markup, any block top or deck between these heights with room to stand
# on. Under vault_max at speed it is gone over instead of climbed, and the speed is kept.
mantle_min 0.55    # below this the feet just step up (LEVEL_STEP_UP)
mantle_max 2.2
vault_max 1.2
roll_fall 4.0      # metres of drop that lands as a roll instead of a stop
wallrun_time 1.2   # seconds a wall run can hold; 0 turns wall running off

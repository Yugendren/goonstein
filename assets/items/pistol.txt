# pistol.txt -- sidearm found wherever the boat keeps its "emergencies".
# Poly Haven photoscan "Service Pistol" (CC0, was assets/models/polyhaven/service_pistol/
# service_pistol_1k.gltf), processed with tools/blender/weapon_prep.py into service_pistol.glb.
# The source file is actually two complete guns laid out side by side (variant "_a" and "_b") plus
# two loose magazines and a loose bullet; kept only pistol_a/slide_a/hammer_a/trigger_a (7556 tris),
# dropped everything else. Reoriented so the muzzle points glTF +Z and up is glTF +Y (barrel was
# +X, up was +Z in the raw scan). Origin moved to the middle of the backstrap/grip (raw x=-0.024,
# z=-0.015, the model's own frame before reorientation) so the model is already in the hand's
# frame -- grip below is therefore 0 0 0. Kept-mesh bounds are 0.032 x 0.130 x 0.222 m
# (X x Y=up x Z=aim); a real service pistol is ~0.21 m, this scan measures 0.222 m, close enough
# that scale stays at 1.0. half below is half of those bounds, same convention as every other item.
model      models/polyhaven/service_pistol/service_pistol.glb
display    Pistol, Safety Allegedly On
weapon     gun
damage     34                          # of a 100-point wind pool
rate       4.0                         # shots per second
range      45                          # metres; hitscan
ammo       12                          # rounds in a full load
knock      3.0                         # m/s shove on whatever it hits
fire_sound shot                        # firing sound
mass       1.0                         # kg
half       0.016 0.065 0.111           # half the model's own bounds (0.032 x 0.130 x 0.222 m)
fragile    0                           # unbreakable
value      0                           # dollars; not evidence of anything
grip       0 -0.02 0   -6 0 0          # lowers it a touch and swings the muzzle left, onto the crosshair
two_handed no
scale      1.0                         # the scan is already real-world size
tint       1.0 1.0 1.0                 # the scan carries its own (photoreal) paint

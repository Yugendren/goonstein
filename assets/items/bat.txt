# bat.txt -- a rounders bat, found leaning by the bunkhouse door. Nobody claims it.
# No CC0 baseball bat model turned up (Quaternius' medieval weapons pack only ships from a Google
# Drive folder with no direct download link; see ASSETS.md), so this is built in-repo from shape
# primitives: models/own/bat.part, 9 pieces, grip at the part's origin and the barrel along +Z.
# Part bounds are 0.07 x 0.07 x 0.873 m (X x Y x Z) with the origin 0.015 m short of the butt end,
# not centred -- half below is half of the part's own total size on each axis, same convention as
# every other shape-built item, even though the box then reaches a little behind the grip that the
# model itself does not.
model      models/own/bat.part
display    Bat, League Unknown
weapon     melee                       # swings, does not shoot
damage     55                          # of a 100-point wind pool
rate       1.6                         # swings per second
range      1.6                         # metres; fixed melee arc
knock      7.0                         # m/s shove on whatever it hits
fire_sound whoosh                      # swing sound
mass       1.2                         # kg
half       0.035 0.035 0.437           # half the part's own bounds (0.07 x 0.07 x 0.873 m)
fragile    0                           # unbreakable
value      0                           # dollars; not evidence of anything
two_handed no
scale      1.0
tint       1.0 1.0 1.0                 # the part's pieces carry their own paint

# crate.txt -- heavy, cheap, built to survive being dropped down a stairwell.

# full crate variants (mirrored nodes at x=-0.5 and x=+0.5) and the loader has no way to pick just
# one, so it drew as a doubled, side-by-side pair. wooden_crate_01 is a single crate (body, lid,
# latch) and the previous half also didn't match its source bounds (0.91 was nearly the model's
# full width, not half of it). Raw model bounds are 0.83 x 0.35 x 0.41 m at scale 1.0; half below
# is half of that.

mass       28                          # kg; heavy even empty
half       0.41 0.17 0.20              # half the model's own bounds at scale 1.0

value      40                          # dollars; cheap
two_handed yes
scale      1.0

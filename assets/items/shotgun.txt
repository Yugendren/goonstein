# shotgun.txt -- the loud one, found near the boat's mooring line. One barrel, and a lot of it.
# No CC0 photoscan of a pump shotgun exists without going through a login-walled source, so
# unlike the pistol, bat and wrench this one is ours: a real parametric CAD solid, built the same
# way the island's buildings are, by tools/cad/build_weapons.py (the shotgun() function) and
# tools/cad/weapon_stage.py, out to models/own/shotgun.glb. 3000 triangles, two flat colours
# (blued steel, walnut), no texture. The model's own origin is the grip point -- the wrist of the
# stock, just behind the trigger guard, where the hand closes -- and the muzzle points +Z in the
# model's own (glTF) axes. Bounds are 0.054 x 0.174 x 1.020 m (X x Y x Z); half below is half of
# those, with the origin off-centre on Y and Z (it sits at the grip, not the middle) and Z reaching
# well behind it into the buttstock, the same way bat.txt's box reaches behind its own grip.
model      models/own/shotgun.glb
display    Shotgun, Duck Season Unclear
weapon     gun
damage     100                         # of a 100-point wind pool; puts a goon on the floor
rate       0.9                         # shots per second
range      18                          # metres; hitscan
ammo       6                           # rounds in a full load
pellets    6                           # hitscan rays per shot, on a fixed fan; a spread, not a bullet
knock      9.0                         # m/s shove on whatever it hits
fire_sound boom                        # firing sound
mass       3.6                         # kg
half       0.027 0.087 0.510           # half the model's own bounds (0.054 x 0.174 x 1.020 m)
fragile    0                           # unbreakable
value      0                           # dollars; not evidence of anything
grip       0.06 -0.10 -0.14   -6 -8 0  # shifts right/down and closer to fill the frame, muzzle swung toward centre
two_handed no
scale      1.0
tint       1.0 1.0 1.0                 # the model carries its own paint
